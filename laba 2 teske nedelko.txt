using System;

class Program
{
    static void Main()
    {
        Console.Write("введи номер фигуры N (1-4): ");
        int N = int.Parse(Console.ReadLine());
        if (N == 1)
        {
            Console.Write("введи радиус круга: ");
            double R = double.Parse(Console.ReadLine());
            double area = Math.PI * R * R;
            Console.WriteLine($"площадь круга: {area}");
        }
        else if (N == 2)
        {
            Console.Write("введи радиус шара: ");
            double R = double.Parse(Console.ReadLine());
            double volume = 4.0 / 3 * Math.PI * Math.Pow(R, 3);
            Console.WriteLine($"объём шара: {volume}");
        }
        else if (N == 3)
        {
            Console.Write("введи радиус цилиндра: ");
            double R = double.Parse(Console.ReadLine());
            Console.Write("введи высоту цилиндра: ");
            double h = double.Parse(Console.ReadLine());
            double volume = Math.PI * R * R * h;
            Console.WriteLine($"объём цилиндра: {volume}");
        }
        else if (N == 4)
        {
            Console.Write("введи радиус сферы: ");
            double r = double.Parse(Console.ReadLine());
            double surface = 4 * Math.PI * r * r;
            Console.WriteLine($"площадь поверхности сферы: {surface}");
        }
        else
        {
            Console.WriteLine("некорректный номер фигуры");
        }
    }
}

        double y = double.Parse(Console.ReadLine());
        Console.Write("Введите z: ");
        double z = double.Parse(Console.ReadLine());
        double u =
            Math.Cos(Math.Abs(x + Math.Pow(Math.Abs(y), 1.0 / 3)))
            + Math.Sin(z - Math.Pow(x, 10) / (1 + Math.Pow(y, 2) + Math.Pow(z, 2)));
        double v = Math.Exp(-Math.Pow(x, 2)) + Math.Pow(Math.Abs(x + y + z), 1.0 / 7) + Math.Exp(y);
        double t;
        if (Math.Pow(u, 2) + Math.Pow(v, 2) >= 5)
        {
            t = Math.Log(v, 2) + Math.Pow(u, 2) + Math.Pow(5, 1.0 / 5);
        }
        else
        {
            t = Math.Log(v) + Math.Pow(u, 2.0 / 3) + Math.Sin(Math.Sqrt(5));
        }

        Console.WriteLine($"u = {u}");
        Console.WriteLine($"v = {v}");
        Console.WriteLine($"t = {t}");
    }

using System;

class Program
{
    static void Main()
    {
        Console.Write("введи x: ");
        double x = double.Parse(Console.ReadLine());
        Console.Write("введи y: ");
        double y = double.Parse(Console.ReadLine());

        double f;
        if (Math.Abs(x) >= Math.Abs(y))
        {
            f = Math.Log(Math.Abs(x + y)) / (Math.Log(Math.Abs(x)) + Math.Log(Math.Abs(y)))
                - Math.Sin(Math.Pow(Math.Pow(x, 8) + Math.Pow(y, 4), 1.0 / 5));
        }
        else
        {
            f = Math.Asin(Math.Abs(x) / Math.Sqrt(Math.Pow(x, 2) + Math.Pow(y, 2)))
                + Math.Exp(Math.Cos(Math.Sqrt(Math.Abs(x + y))));
        }

        double pow8sum = Math.Pow(x, 8) + Math.Pow(y, 8);
        double u;
        if (pow8sum >= 12)
        {
            u = Math.Sin(Math.Abs(x - y / pow8sum));
        }
        else
        {
            u = Math.Cos(Math.Abs(x - y / pow8sum));
        }

        Console.WriteLine($"f = {f}");
        Console.WriteLine($"u = {u}");
    }
}
