using System;
class Program
{
    static void Main()
    {
    Console.WriteLine("введи а");
    int a=int.Parse(Console.ReadLine());
    Console.WriteLine("введи b");
    int b=int.Parse(Console.ReadLine());
    long sum=0;
    int count=0;
    for(int i=a; i<=b; i++)
        {
            sum+=i;
            count++;
        }
    double avg=(double)sum/count;
    Console.WriteLine($"sum={sum}     count={count}");
    Console.WriteLine($"avg {avg}");
    }
}

**zadanie 13**

using System;
namespace sosedi
{
    class Program
    {
        static void Main()
        {
            Console.WriteLine("дроби вводите через запятую!");
            Console.WriteLine("введите количество чисел n:");
            int n = Convert.ToInt32(Console.ReadLine());
            int k = 0;
            double prev = 0;
            double cur = 0;
            for (int i = 1; i <= n; i++)
            {
                double next;
                do
                {
                    Console.WriteLine("введите число " + i + ":");
                    next = Convert.ToDouble(Console.ReadLine());
                }
                while (next < 0);
                if (i >= 3 && cur > prev && cur > next)
                {
                    k++;
                }
                prev = cur;
                cur = next;
            }
            Console.WriteLine("чисел, больших обоих соседей: " + k);
        }
    }
}

using System;
namespace znak
{
    class Program
    {
        static void Main()
        {
            Console.WriteLine("введите количество чисел n:");
            int n = Convert.ToInt32(Console.ReadLine());
            int k = 0;
            int prev = 0;
            for (int i = 1; i <= n; i++)
            {
                int x;
                do
                {
                    Console.WriteLine("введите число " + i + ":");
                    x = Convert.ToInt32(Console.ReadLine());
                }
                while (x == 0);
                if (i > 1 && (x > 0) != (prev > 0))
                {
                    k++;
                }
                prev = x;
            }
            Console.WriteLine("количество смен знака: " + k);
        }
    }
}

using System;
namespace sovershenoe
{
    class Program
    {
        static void Main()
        {
            Console.WriteLine("введите натуральное число:");
            int n = Convert.ToInt32(Console.ReadLine());
            int s = 0;
            for (int d = 1; d <= n / 2; d++)
            {
                if (n % d == 0)
                {
                    s = s + d;
                }
            }
            if (n > 0 && s == n)
            {
                Console.WriteLine(n + " - совершенное число");
            }
            else
            {
                Console.WriteLine(n + " - не совершенное число (сумма делителей = " + s + ")");
            }
        }
    }
}
