using System;
namespace moloko
{
    class Program
    {
        static void Main()
        {
            Console.WriteLine("дроби вводите через запятую!");
            Console.WriteLine("введите количество фирм n:");
            int n = Convert.ToInt32(Console.ReadLine());
            double min = 1000000000;
            int nom = 0;
            for (int i = 1; i <= n; i++)
            {
                Console.WriteLine("фирма " + i);
                Console.WriteLine("введите x1:");
                double x1 = Convert.ToDouble(Console.ReadLine());
                Console.WriteLine("введите y1:");
                double y1 = Convert.ToDouble(Console.ReadLine());
                Console.WriteLine("введите z1:");
                double z1 = Convert.ToDouble(Console.ReadLine());
                Console.WriteLine("введите x2:");
                double x2 = Convert.ToDouble(Console.ReadLine());
                Console.WriteLine("введите y2:");
                double y2 = Convert.ToDouble(Console.ReadLine());
                Console.WriteLine("введите z2:");
                double z2 = Convert.ToDouble(Console.ReadLine());
                Console.WriteLine("введите c1 (через запятую):");
                double c1 = Convert.ToDouble(Console.ReadLine());
                Console.WriteLine("введите c2 (через запятую):");
                double c2 = Convert.ToDouble(Console.ReadLine());
                double s1 = 2 * (x1 * y1 + x1 * z1 + y1 * z1);
                double s2 = 2 * (x2 * y2 + x2 * z2 + y2 * z2);
                double v1 = x1 * y1 * z1;
                double v2 = x2 * y2 * z2;
                double d = s1 * v2 - s2 * v1;
                double m = (s1 * c2 - s2 * c1);
                double lit = m * 1000;
                if (lit < min)
                {
                    min = lit;
                    nom = i;
                }
            }
            Console.WriteLine("{0} {1:F2}", nom, min);
        }
    }
}
