using System;
namespace posled
{
    class Program
    {
        static void Main()
        {
            Console.WriteLine("вводите числа, ноль - конец последовательности:");
            int x = Convert.ToInt32(Console.ReadLine());
            long s = 0;
            int k = 0;
            while (x != 0)
            {
                int m = Math.Abs(x);
                int sd = 0;
                bool dva = false;
                while (m > 0)
                {
                    sd = sd + m % 10;
                    if (m % 10 == 2)
                    {
                        dva = true;
                    }
                    m = m / 10;
                }
                bool pr = sd >= 2;
                for (int d = 2; d * d <= sd; d++)
                {
                    if (sd % d == 0)
                    {
                        pr = false;
                    }
                }
                if (pr)
                {
                    s = s + x;
                }
                if (x > 0 && dva)
                {
                    k++;
                }
                x = Convert.ToInt32(Console.ReadLine());
            }
            Console.WriteLine("а) сумма чисел, у которых сумма цифр простая: " + s);
            Console.WriteLine("б) положительных чисел с цифрой 2: " + k);
        }
    }
}

    {

            Console.WriteLine("вводите неотрицательные числа, отрицательное - конец ряда:");

            double prev = 0;
            double cur = 0;
            int i = 0;
            int k = 0;
            while (next >= 0)
            {
                i++;
                if (i >= 3 && cur > prev && cur > next)
                {
                    k++;

                prev = cur;

                next = Convert.ToDouble(Console.ReadLine());
            }
            Console.WriteLine("чисел, больших обоих соседей: " + k);
        }
    }

            Console.WriteLine("дроби вводите через запятую!");

using System;
namespace sinus
{
    class Program
    {
        static void Main()
        {
            const double eps = 0.00001;
            Console.WriteLine("дроби вводите через запятую!");
            Console.WriteLine("введите x:");
            double x = Convert.ToDouble(Console.ReadLine());
            double y = x;
            double t = x;
            int k = 1;
            while (Math.Abs(t) >= eps)
            {
                t = -t * x * x / ((2 * k) * (2 * k + 1));
                y = y + t;
                k++;
            }
            Console.WriteLine("ряд: y = {0:F6}", y);
            Console.WriteLine("sin(x) = {0:F6}", Math.Sin(x));
        }
    }
}
